27def _homogeneous(R: Any, p: Any, last: float = 1.0) -> Any: 

28 b = get_backend() 

29 top = b.concatenate((R, p.reshape(3, 1)), axis=1)

33def transform_from_twist(S, theta: float) -> NDArray: 

34 """Compute an SE(3) transform from a normalized screw and displacement.""" 

35 b = get_backend() 

36 S = b.asarray(S) 

37 omega, v = S[:3], S[3:] 

38 omega_hat = skew_symmetric(omega) 

39 omega_hat2 = b.matmul(omega_hat, omega_hat) 

40 R = b.eye(3) + b.sin(theta) * omega_hat + (1 - b.cos(theta)) * omega_hat2 

41 G = b.eye(3) * theta + (1 - b.cos(theta)) * omega_hat + (theta - b.sin(theta)) * omega_hat2 

42 return _homogeneous(R, b.matmul(G, v))

45def adjoint_transform(T) -> NDArray: 

46 """Compute the 6x6 adjoint of a homogeneous transform.""" 

47 b = get_backend() 

48 T = b.asarray(T) 

49 R, p = T[:3, :3], T[:3, 3] 

50 zero = b.zeros((3, 3), dtype=R.dtype) 

51 return b.concatenate((b.concatenate((R, zero), axis=1), 

52 b.concatenate((b.matmul(skew_symmetric(p), R), R), axis=1)), axis=0)

55def _log6_phi_sq_coeff(b: Any, theta_sq: Any) -> Any: 

56 """Return ``a = (1 - (theta/2) cot(theta/2)) / theta^2``, with ``a(0) = 1/12``. 

57 

58 This is the ``Phi^2`` coefficient of ``theta * G^-1`` when the rotational 

59 logarithm is carried as the matrix ``Phi = theta * w_hat`` rather than as a 

60 separate unit axis and angle. Written as an even function of ``theta`` 

61 (smooth in ``theta^2``, so no ``sqrt`` feeds the graph near zero); the exact 

62 form is a vanishing/vanishing ratio as ``theta -> 0``, so below 

63 ``theta^2 < 1e-2`` (``theta < 0.1``) the four-term series 

64 ``1/12 + theta^2/720 + theta^4/30240 + theta^6/1209600`` is used instead. 

65 

66 That band is deliberately WIDER than the ``theta^2 < 1e-4`` one 

67 ``_exp6_trans_coeffs`` uses, because this coefficient's exact form is a 

68 double cancellation: ``1 - cos theta`` loses about five digits at 

69 ``theta = 1e-2``, and the numerator ``1 - theta sin theta / (2 (1 - cos 

70 theta))`` is itself ``theta^2/12`` recovered by subtracting two nearly equal 

71 quantities. Its *derivative* is degraded well before its value is, so 

72 switching at ``1e-2`` handed the coefficient's gradient to a branch that had 

73 already lost roughly four percent of it. Four series terms hold the Taylor 

74 side to ~1e-13 relative accuracy out to ``theta = 0.1``, by which point the 

75 exact form has recovered. 

76 

77 The clamp inside the exact branch sits strictly BELOW the switch point: 

78 clamping at the switch value makes ``maximum`` tie exactly where the exact 

79 branch becomes active, and autodiff splits a tie's derivative in half. 

80 """ 

81 small = ( 

82 1 / 12 

83 + theta_sq / 720 

84 + theta_sq**2 / 30240 

85 + theta_sq**3 / 1209600 

86 ) 

87 theta = b.sqrt(b.maximum(theta_sq, b.asarray(5e-3))) 

88 exact = ( 

89 1 - theta * b.sin(theta) / b.maximum(2 * (1 - b.cos(theta)), b.asarray(1e-300)) 

90 ) / (theta * theta) 

91 return b.where(theta_sq < 1e-2, small, exact)

94def _theta_g_inverse(b: Any, phi: Any) -> Any: 

95 """Return ``theta * G^-1``, smooth and finite through zero. 

96 

97 ``G^-1`` alone diverges like ``1/theta``, but the product both logarithms 

98 actually need does not: it is ``I - Phi/2 + a(theta^2) Phi^2`` and tends to 

99 the identity, which is why the pure-translation result falls out of the 

100 general formula rather than needing a separate branch. 

101 

102 Everything is derived from ``Phi = MatrixLog3(R)``, including 

103 ``theta^2 = |Phi|_F^2 / 2``. Taking the axis and angle from 

104 ``rotation_logm`` instead would reintroduce its thresholded axis, which is 

105 zeroed below theta = 1e-6 and would flatten this whole expression to the 

106 identity there -- losing the ``-Phi/2`` term that carries the translational 

107 correction and its derivative. 

108 """ 

109 theta_sq = b.sum(phi * phi) / 2 

110 coeff = _log6_phi_sq_coeff(b, theta_sq) 

111 return b.eye(3) - 0.5 * phi + coeff * b.matmul(phi, phi)

114def logm(T) -> NDArray: 

115 """Return the six-vector logarithm of a homogeneous transform.""" 

116 b = get_backend() 

117 T = b.asarray(T) 

118 p = T[:3, 3] 

119 # Everything derives from Phi = MatrixLog3(R), for both blocks. 

120 # ``rotation_logm`` zeroes its axis below theta = 1e-6 (the axis is genuinely 

121 # undefined at the identity), so any expression built from that axis loses a 

122 # small-but-real rotation and its derivative at the origin. MatrixLog3 

123 # evaluates 0.5 (theta/sin theta) (R - R.T) through a Taylor-safe 

124 # coefficient and is smooth in value and gradient there. 

125 # 

126 # No small-angle branch on the translation either: ``theta * G^-1`` -> I as 

127 # theta -> 0, so the general formula already yields the pure-translation 

128 # result that the removed ``where`` used to select. 

129 phi = MatrixLog3(T[:3, :3]) 

130 rotvec = skew_symmetric_to_vector(phi) 

131 return b.concatenate((rotvec, b.matmul(_theta_g_inverse(b, phi), p)))

134def se3ToVec(se3_matrix) -> NDArray: 

135 """Convert an se(3) matrix to a six-vector.""" 

136 b = get_backend() 

137 se3_matrix = b.asarray(se3_matrix) 

138 if se3_matrix.shape != (4, 4): 

139 raise ValueError("Input matrix must be a 4x4 matrix.") 

140 omega = b.stack((se3_matrix[2, 1], se3_matrix[0, 2], se3_matrix[1, 0])) 

141 return b.concatenate((omega, se3_matrix[:3, 3]))

149def TransInv(T) -> NDArray: 

150 """Invert a homogeneous transform.""" 

151 b = get_backend() 

152 T = b.asarray(T) 

153 R, p = TransToRp(T) 

154 Rt = R.T 

155 return _homogeneous(Rt, -b.matmul(Rt, p))

158def MatrixLog6(T) -> NDArray: 

159 """Compute the se(3) matrix logarithm of a homogeneous transform.""" 

160 b = get_backend() 

161 T = b.asarray(T) 

162 R, p = TransToRp(T) 

163 # See ``logm``: both blocks derive from Phi = MatrixLog3(R), which is smooth 

164 # through the identity, and the general translation formula already 

165 # degenerates to the pure-translation result at theta = 0, so branching on a 

166 # small angle only served to discard the rotation and its gradient. 

167 phi = MatrixLog3(R) 

168 v = b.matmul(_theta_g_inverse(b, phi), p) 

169 return _homogeneous(phi, v, last=0.0)

172def _exp6_trans_coeffs(b: Any, theta_sq: Any) -> Tuple[Any, Any]: 

173 """Return the SE(3) translation coefficients ``((1-cos th)/th^2, (th-sin th)/th^3)``. 

174 

175 Both are even functions of ``theta`` (smooth in ``theta^2``, so no ``sqrt`` 

176 feeds the graph near zero) and therefore have a finite gradient through 

177 ``theta = 0``. Near the identity (``theta^2 < 1e-4``, i.e. ``theta < 1e-2``) 

178 the Taylor series is used; elsewhere the exact form with ``theta`` clamped 

179 away from zero so the inactive branch's gradient stays finite. The band is set 

180 by where the exact forms' *backward* pass stops cancelling catastrophically 

181 (the ``(th - sin th)/th^3`` gradient error is ~128 at ``theta = 1e-6`` and only 

182 falls below ~1e-10 near ``theta ~ 1e-2``), not merely where the primal is 

183 finite; the too-small ``1e-12`` cutoff left the exact backward corrupt just 

184 above it. 

185 """ 

186 near_zero = theta_sq < 1e-4 

187 theta_safe = b.sqrt(b.maximum(theta_sq, b.asarray(1e-12))) 

188 c1_exact = (1 - b.cos(theta_safe)) / (theta_safe * theta_safe) 

189 c2_exact = (theta_safe - b.sin(theta_safe)) / (theta_safe * theta_safe * theta_safe) 

190 c1_taylor = 0.5 - theta_sq / 24.0 + theta_sq * theta_sq / 720.0 

191 c2_taylor = 1.0 / 6.0 - theta_sq / 120.0 + theta_sq * theta_sq / 5040.0 

192 return b.where(near_zero, c1_taylor, c1_exact), b.where(near_zero, c2_taylor, c2_exact)

195def MatrixExp6(se3mat) -> NDArray: 

196 """Compute the SE(3) exponential of an se(3) matrix.""" 

197 b = get_backend() 

198 se3mat = b.asarray(se3mat) 

199 if se3mat.shape != (4, 4): 

200 raise ValueError("Input matrix must be of shape (4, 4)") 

201 omega_theta = b.stack((se3mat[2, 1], se3mat[0, 2], se3mat[1, 0])) 

202 v = se3mat[:3, 3] 

203 K = se3mat[:3, :3] 

204 theta_sq = ( 

205 omega_theta[0] * omega_theta[0] 

206 + omega_theta[1] * omega_theta[1] 

207 + omega_theta[2] * omega_theta[2] 

208 ) 

209 # Rotational part reuses the (locally smooth) SO(3) exponential; translation 

210 # is p = [I + C1*K + C2*K^2] v with C1 = (1-cos th)/th^2, C2 = (th-sin th)/th^3. 

211 # Folding the small-angle branch into the Taylor-safe coefficients (instead of 

212 # a where onto a constant) keeps dT/dtheta correct and finite at theta = 0. 

213 coeff_c1, coeff_c2 = _exp6_trans_coeffs(b, theta_sq) 

214 k_v = b.matmul(K, v) 

215 p = v + coeff_c1 * k_v + coeff_c2 * b.matmul(K, k_v) 

216 return _homogeneous(MatrixExp3(K), p)
