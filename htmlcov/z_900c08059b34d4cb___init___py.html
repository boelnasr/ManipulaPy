32# Re-export the implementation module's full namespace (public API plus the 

33# single-underscore helpers that existing callers/tests import) so that every name 

34# that used to live in the flat ManipulaPy.path_planning module stays importable 

35# from ManipulaPy.planning. This is done dynamically rather than with an explicit 

36# name list because part of that namespace (e.g. MockCuda and the CUDA-only kernel 

37# symbols) is defined conditionally on CUDA availability, so a hard-coded list 

38# would break imports on machines with a different CUDA configuration. Dunder names 

39# are skipped so this package keeps its own module machinery. 

40globals().update( 

41 { 

42 name: value 

43 for name, value in vars(trajectory_planning).items() 

44 if not name.startswith("__") 

45 } 

46) 

47 

48# The trajectory planning implementation is split across internal helper modules 

49# (_kernels, trajectory, trajectory_dynamics, collision_host, _plotting). Importing 

50# trajectory_planning above pulls those submodules in and the import system binds 

51# them as attributes of this package. They are implementation detail, so drop them 

52# here to keep the historical public namespace unchanged. 

53for _internal in ( 

54 "_kernels", 

55 "trajectory", 

56 "trajectory_dynamics", 

57 "collision_host", 

58 "_plotting", 

59 "benchmarks", 

60 "_runtime", 

61 "_sys", 

62 "_ModuleType", 

63 "_FORWARDED_RUNTIME_NAMES", 

64 "_PlanningCompatibilityModule", 

65): 

66 globals().pop(_internal, None) 

67del _internal 

68 

69__all__ = list(trajectory_planning.__all__)
