3""" 

4PyTorch array backend - ManipulaPy 

5 

6:class:`ArrayBackend` implementation backed by PyTorch. This module imports 

7``torch`` at the top level and therefore must never be imported eagerly: the 

8backend package registers it lazily behind an import probe (see 

9``ManipulaPy/backend/__init__.py``) so that ManipulaPy remains importable on 

10machines with only NumPy installed. 

11 

12Unlike the NumPy and CuPy backends, PyTorch tensors can carry autograd / trace 

13state, so this backend reports ``is_concrete = False``. That gates the 

14value-keyed caches (e.g. the mass-matrix cache) off for traced inputs, keeping 

15gradients attached rather than detaching them through a host-synced cache key. 

16It defaults to the CPU device and ``gpu_capable = False`` so a CPU-default Torch 

17backend never routes numeric work to the Numba CUDA kernels; the numerics match 

18NumPy because floating-point construction is forced to ``float64`` on CPU.

46# --- NumPy <-> Torch dtype bridge ------------------------------------------ 

47# Bidirectional dtype map so promotion can be delegated to ``numpy.result_type`` 

48# and the answer mapped back to a Torch dtype. Torch 2.7 exposes uint16/uint32/ 

49# uint64/bool/complex dtypes, but many kernels (``eye``, ``matmul``, ``amax``, 

50# ``maximum`` on complex, ...) still reject them while NumPy succeeds. Rather 

51# than special-case each op, those exotic dtypes are routed through NumPy (see 

52# ``TorchBackend._numpy_fallback``) so the result matches NumPy by construction. 

53_NP_TO_TORCH_DTYPE = { 

54 np.bool_: torch.bool, 

55 np.int8: torch.int8, 

56 np.int16: torch.int16, 

57 np.int32: torch.int32, 

58 np.int64: torch.int64, 

59 np.uint8: torch.uint8, 

60 np.uint16: torch.uint16, 

61 np.uint32: torch.uint32, 

62 np.uint64: torch.uint64, 

63 np.float16: torch.float16, 

64 np.float32: torch.float32, 

65 np.float64: torch.float64, 

66 np.complex64: torch.complex64, 

67 np.complex128: torch.complex128, 

68} 

69_TORCH_TO_NP_DTYPE = {t: n for n, t in _NP_TO_TORCH_DTYPE.items()} 

70# Exotic Torch dtypes whose kernels do not (fully) match NumPy: bool, the 

71# unsigned integers, and complex. Operations on these dtypes are delegated to 

72# NumPy so the value *and* dtype equal NumPy's own result. They never carry 

73# autograd/trace state and never appear on ManipulaPy's float64 hot paths, so 

74# the ``.detach().cpu().numpy()`` round-trip is safe. 

75_FALLBACK_DTYPES = frozenset( 

76 { 

77 torch.bool, 

78 torch.uint8, 

79 torch.uint16, 

80 torch.uint32, 

81 torch.uint64, 

82 torch.complex64, 

83 torch.complex128, 

84 } 

85)

88def _torch_to_np_dtype(dtype: "torch.dtype") -> "np.dtype": 

89 """Map a Torch dtype to the equivalent NumPy dtype.""" 

90 return np.dtype(_TORCH_TO_NP_DTYPE[dtype])

93def _np_to_torch_dtype(dtype: Any) -> "torch.dtype": 

94 """Map a NumPy dtype (or dtype-like) to the equivalent Torch dtype.""" 

95 return _NP_TO_TORCH_DTYPE[np.dtype(dtype).type]

98def _np_result_dtype(*operands: Any) -> "torch.dtype": 

99 """Torch dtype NumPy would produce by promoting ``operands`` together. 

100 

101 Promotion is delegated to ``numpy.result_type`` (NEP 50 rules): tensors and 

102 NumPy scalars contribute their dtype (array/strong semantics) while Python 

103 scalars contribute weak-scalar semantics, so the mapped Torch dtype matches 

104 what the NumPy backend returns for the same operands. ``None`` operands 

105 (e.g. an omitted clip bound) are skipped. 

106 """ 

107 np_args = [] 

108 for op in operands: 

109 if op is None: 

110 continue 

111 if isinstance(op, torch.Tensor): 

112 np_args.append(_torch_to_np_dtype(op.dtype)) 

113 elif isinstance(op, np.generic): 

114 np_args.append(op.dtype) 

115 elif isinstance(op, (bool, int, float, complex)): 

116 np_args.append(op) 

117 else: 

118 np_args.append(np.asarray(op).dtype) 

119 return _np_to_torch_dtype(np.result_type(*np_args))

122def _as_torch_dtype(dtype: Any) -> "torch.dtype": 

123 """Normalize a dtype argument to a ``torch.dtype``. 

124 

125 Accepts a ``torch.dtype`` unchanged and maps NumPy dtype forms 

126 (``np.float32``, ``np.dtype("float32")``, ``"float32"``) through the bridge 

127 so the four constructors honour NumPy-style dtype arguments like NumPy does. 

128 """ 

129 if isinstance(dtype, torch.dtype): 

130 return dtype 

131 return _np_to_torch_dtype(dtype)

134def _np_reduce_dtype(t: "torch.Tensor") -> "torch.dtype": 

135 """Torch dtype of NumPy's additive accumulator (``sum``/``trace``) for ``t``. 

136 

137 NumPy upcasts bool/int8/int16/int32 to int64 and uint8/uint16/uint32 to 

138 uint64 while preserving int64/uint64/float*/complex*; this delegates that 

139 rule to ``numpy.sum`` on a zero-size array of the same dtype. 

140 """ 

141 np_dtype = np.zeros((), dtype=_torch_to_np_dtype(t.dtype)).sum().dtype 

142 return _np_to_torch_dtype(np_dtype)

145class TorchBackend(ArrayBackend): 

146 """Backend backed by PyTorch tensors. 

147 

148 Defaults to the CPU device with ``float64`` floating-point construction so 

149 numerics match the NumPy backend. Set ``device="cuda"`` to materialize 

150 tensors on the GPU; ``gpu_capable`` stays ``False`` regardless because the 

151 Numba CUDA kernel routing is a separate acceleration path. 

152 """ 

153 

154 float32 = torch.float32 

155 float64 = torch.float64 

156 #: Torch tensors can carry autograd/trace state and are not hashable by 

157 #: value, so value-keyed caches (e.g. the mass-matrix cache) must be 

158 #: bypassed rather than keyed by tensor identity or a host-synced value. 

159 is_concrete = False 

160 #: CPU-default Torch backend: never routes to the Numba CUDA kernel path. 

161 gpu_capable = False 

162 

163 def __init__(self, device: str = "cpu") -> None: 

164 self._device = torch.device(device) 

165 

166 def cache_token(self) -> Any: 

167 """Namespace cached arrays to this backend instance and device.""" 

168 return (self, str(self._device))

170 # -- construction --------------------------------------------------- 

171 @staticmethod 

172 def _contains_tensor(obj: Any) -> bool: 

173 """True iff ``obj`` is, or nests, a ``torch.Tensor`` leaf.""" 

174 if isinstance(obj, torch.Tensor): 

175 return True 

176 if isinstance(obj, (list, tuple)): 

177 return any(TorchBackend._contains_tensor(el) for el in obj) 

178 return False 

179 

180 @staticmethod 

181 def _native_byteorder(arr: "np.ndarray") -> "np.ndarray": 

182 """Return ``arr`` in native byte order (``torch.from_numpy`` requires it).""" 

183 if arr.dtype.byteorder not in ("=", "|"): 

184 return arr.astype(arr.dtype.newbyteorder("=")) 

185 return arr 

186 

187 def _to_tensor(self, obj: Any) -> "torch.Tensor": 

188 """Convert ``obj`` to a tensor, preserving tensor inputs (and autograd). 

189 

190 A (possibly nested) sequence containing tensor leaves is assembled with 

191 ``torch.stack`` so autograd/trace state stays attached -- this is the 

192 build-matrix-from-computed-trig pattern in kinematics. The sequence's 

193 joint dtype is computed with weak-scalar semantics (Python scalars weak, 

194 tensors strong) so ``[int64_tensor, 2]`` stays int64 and 

195 ``[int64_tensor, float32_tensor]`` promotes to float64, matching NumPy. 

196 Pure Python/NumPy input routes through NumPy so Python floats become 

197 ``float64`` (matching NumPy) instead of Torch's ``float32`` default. 

198 """ 

199 if isinstance(obj, torch.Tensor): 

200 return obj 

201 if isinstance(obj, np.ndarray): 

202 obj = self._native_byteorder(obj) 

203 if any(stride < 0 for stride in obj.strides): 

204 obj = np.ascontiguousarray(obj) 

205 return torch.from_numpy(obj) 

206 if isinstance(obj, (list, tuple)) and self._contains_tensor(obj): 

207 # Co-locate every leaf on this backend's device and cast it to the 

208 # sequence's NumPy-joint dtype before stacking: a CUDA-device 

209 # backend may stack a CUDA tensor leaf with a Python-scalar leaf 

210 # built on the CPU. ``.to(...)`` preserves autograd, and dtype/device 

211 # casts that are already satisfied (the all-float64 trig path) are 

212 # no-ops that keep the graph attached. 

213 target = self._seq_result_dtype(obj) 

214 return torch.stack( 

215 [self._stack_leaf(el).to(target).to(self._device) for el in obj] 

216 ) 

217 return torch.as_tensor(np.asarray(obj)) 

218 

219 def _seq_result_dtype(self, obj: Any) -> "torch.dtype": 

220 """Joint Torch dtype for a (possibly nested) tensor-containing sequence. 

221 

222 Leaves are gathered recursively so ``numpy.result_type`` sees each 

223 contributor with weak/strong semantics (Python scalars weak, tensors 

224 strong), matching what ``np.array`` would produce for the same nesting. 

225 """ 

226 contributors: list = [] 

227 

228 def _walk(node: Any) -> None: 

229 if isinstance(node, (list, tuple)): 

230 for element in node: 

231 _walk(element) 

232 else: 

233 contributors.append(node) 

234 

235 _walk(obj) 

236 return _np_result_dtype(*contributors) 

237 

238 def _stack_leaf(self, el: Any) -> "torch.Tensor": 

239 """Convert one element of a tensor-containing sequence to a tensor. 

240 

241 Tensors and nested sequences defer to :meth:`_to_tensor`; Python scalars 

242 are materialized (as ``float64``) and cast to the sequence's joint dtype 

243 by the caller. 

244 """ 

245 if isinstance(el, (torch.Tensor, list, tuple, np.ndarray)): 

246 return self._to_tensor(el) 

247 return torch.as_tensor(el, dtype=torch.float64)

250 @classmethod 

251 def _needs_fallback(cls, *tensors: Any) -> bool: 

252 """True iff any tensor operand has an exotic (NumPy-delegated) dtype.""" 

253 return any( 

254 isinstance(t, torch.Tensor) and t.dtype in _FALLBACK_DTYPES 

255 for t in tensors 

256 )

258 @staticmethod 

259 def _is_python_int(value: Any) -> bool: 

260 """True for a Python ``int`` (weak scalar), excluding ``bool``.""" 

261 return isinstance(value, int) and not isinstance(value, bool) 

262 

263 @staticmethod 

264 def _as_numpy_operand(op: Any) -> Any: 

265 """View a tensor operand as NumPy; pass scalars/None/array-likes through. 

266 

267 Detaching to a host NumPy view is safe here because the fallback only 

268 runs for non-float, torch-unsupported, or overflow cases, which never 

269 carry autograd/trace state. 

270 """ 

271 if isinstance(op, torch.Tensor): 

272 return op.detach().cpu().numpy() 

273 return op 

274 

275 def _from_numpy(self, result: Any) -> "torch.Tensor": 

276 """Materialize a NumPy op result as a tensor on this backend's device.""" 

277 arr = self._native_byteorder(np.asarray(result)) 

278 # ``np.ascontiguousarray`` upgrades a 0-D scalar to shape ``(1,)``; only 

279 # apply it to arrays so scalar reductions (e.g. ``sum(bool)``) keep the 

280 # 0-D shape NumPy returns. 

281 if arr.ndim > 0: 

282 arr = np.ascontiguousarray(arr) 

283 return torch.from_numpy(arr).to(self._device) 

284 

285 def _numpy_fallback(self, np_func: Any, *operands: Any, **kwargs: Any) -> Any: 

286 """Compute ``np_func`` on NumPy views of ``operands`` and return a tensor. 

287 

288 This yields exact NumPy parity by construction: the operands are exactly 

289 what the NumPy backend would receive (tensors as their host views, Python 

290 scalars kept weak), so the value and dtype equal NumPy's own result. 

291 """ 

292 np_args = [self._as_numpy_operand(op) for op in operands] 

293 return self._from_numpy(np_func(*np_args, **kwargs))

295 @staticmethod 

296 def _promote_pair( 

297 a: "torch.Tensor", b: "torch.Tensor" 

298 ) -> Tuple["torch.Tensor", "torch.Tensor"]: 

299 """Cast ``a``/``b`` to the dtype NumPy would promote them to jointly. 

300 

301 Shared by the binary ops (matmul/solve/maximum/minimum/cross) so a 

302 mixed-dtype pair follows ``np.result_type`` instead of Torch's own 

303 promotion (or a raise). Same-dtype pairs -- the all-float64 hot path -- 

304 short-circuit with no ``result_type`` call and no copy. 

305 """ 

306 if a.dtype == b.dtype: 

307 return a, b 

308 target = _np_result_dtype(a, b) 

309 return a.to(target), b.to(target) 

310 

311 @staticmethod 

312 def _promote_float(t: "torch.Tensor") -> "torch.Tensor": 

313 """Promote integer/boolean tensors to ``float64`` to match NumPy. 

314 

315 NumPy's transcendental, ``mean``, and linear-algebra operations return 

316 ``float64`` for integer (or boolean) input, whereas Torch either raises 

317 (``mean``/``norm``/linalg) or silently returns ``float32`` (``sin``, 

318 ``cos``, ``sqrt``, ...). Floating (and complex) tensors pass through 

319 unchanged so a ``float32`` input keeps its precision, exactly as NumPy 

320 preserves it. 

321 """ 

322 if t.is_floating_point() or t.is_complex(): 

323 return t 

324 return t.to(torch.float64)

342 def eye(self, n: int, dtype: Optional[Any] = None) -> Any: 

343 dtype = torch.float64 if dtype is None else _as_torch_dtype(dtype) 

344 # torch.eye rejects several exotic dtypes (uint16/32/64, ...) that NumPy 

345 # constructs; build via NumPy so the result matches np.eye by dtype. 

346 if dtype in _FALLBACK_DTYPES: 

347 return self._from_numpy(np.eye(n, dtype=_torch_to_np_dtype(dtype))) 

348 return torch.eye(n, dtype=dtype, device=self._device)

356 def _promote_operands(self, arrays: Any) -> list: 

357 """Tensorize ``arrays`` and cast them to their joint ``np.result_type``. 

358 

359 np.stack/np.concatenate promote all inputs together (e.g. int64 + float32 

360 -> float64), whereas torch would keep float32; casting to the NumPy 

361 result dtype preserves that precision. 

362 """ 

363 tensors = [self.asarray(a) for a in arrays] 

364 if not tensors: 

365 return tensors 

366 target = _np_result_dtype(*tensors) 

367 return [t if t.dtype == target else t.to(target) for t in tensors]

384 def pinv(self, a: Any) -> Any: 

385 # np.linalg.pinv zeroes singular values <= rcond * largest_sv with a 

386 # default rcond of 1e-15. torch.linalg.pinv's default rtol is far larger 

387 # (max(M, N) * eps), so a tiny sv would be inverted to ~1e15 instead of 

388 # zeroed; pass the NumPy-compatible relative cutoff as rtol to match. 

389 return torch.linalg.pinv(self._promote_float(self.asarray(a)), rtol=1e-15)

391 def solve(self, a: Any, b: Any) -> Any: 

392 # Integer/bool operands upcast to float64 (torch.linalg.solve rejects 

393 # them) and mixed floating precisions promote jointly by np.result_type 

394 # so e.g. solve(float32, int64) -> float64 like NumPy. 

395 a = self._promote_float(self.asarray(a)) 

396 b = self._promote_float(self.asarray(b)) 

397 return torch.linalg.solve(*self._promote_pair(a, b))

406 def trace(self, a: Any) -> Any: 

407 a = self.asarray(a) 

408 # torch.trace rejects bool/unsigned dtypes NumPy accepts; delegate those. 

409 if self._needs_fallback(a): 

410 return self._numpy_fallback(np.trace, a) 

411 target = _np_reduce_dtype(a) 

412 if a.ndim == 2 and a.dtype != torch.float16: 

413 out = torch.trace(a) 

414 else: 

415 # np.trace defaults to axis1=0, axis2=1 for stacked/batched input; 

416 # torch.trace only accepts 2D and rejects float16, so mirror NumPy 

417 # via the diagonal sum (which also handles the batched case). 

418 out = torch.diagonal(a, dim1=0, dim2=1).sum(dim=-1) 

419 return out if out.dtype == target else out.to(target)

422 def _transcendental(self, np_func: Any, torch_func: Any, x: Any) -> Any: 

423 """Apply a float-producing unary op, matching NumPy's dtype promotion. 

424 

425 NumPy promotes integer/boolean input to a *width-dependent* float 

426 (int8/uint8/bool -> float16, int16/uint16 -> float32, ...), which Torch 

427 cannot reproduce, so integer/boolean input is delegated to NumPy. Float 

428 and complex input stay on the native Torch path (autograd-preserving). 

429 """ 

430 x = self.asarray(x) 

431 if not (x.is_floating_point() or x.is_complex()): 

432 return self._numpy_fallback(np_func, x) 

433 return torch_func(x)

447 @staticmethod 

448 def _is_weak_real_scalar(value: Any) -> bool: 

449 """True for a Python ``int``/``float`` weak scalar (excluding ``bool``).""" 

450 return isinstance(value, (int, float)) and not isinstance(value, bool)

452 def arctan2(self, y: Any, x: Any) -> Any: 

453 ty, tx = self.asarray(y), self.asarray(x) 

454 # np.arctan2 promotes both operands jointly to a width-dependent float and 

455 # raises on complex. Stay on the native (autograd-preserving) Torch path 

456 # when the computation is real-float: at least one float tensor, and every 

457 # other operand a float tensor or a weak Python real scalar. Otherwise 

458 # delegate to NumPy for exact parity. 

459 y_float, x_float = ty.is_floating_point(), tx.is_floating_point() 

460 native = ( 

461 (y_float or x_float) 

462 and (y_float or self._is_weak_real_scalar(y)) 

463 and (x_float or self._is_weak_real_scalar(x)) 

464 ) 

465 if not native: 

466 return self._numpy_fallback(np.arctan2, y, x) 

467 return torch.arctan2(*self._promote_weak(y, x))

469 def abs(self, x: Any) -> Any: 

470 x = self.asarray(x) 

471 # torch.abs rejects bool and complex-parity differs; delegate exotics. 

472 if self._needs_fallback(x): 

473 return self._numpy_fallback(np.abs, x) 

474 return torch.abs(x)

476 def clip(self, x: Any, a_min: Any, a_max: Any) -> Any: 

477 xt = self.asarray(x) 

478 # Delegate to NumPy for exotic dtypes and for an integer array bounded by 

479 # an OUT-OF-RANGE weak Python int, where np.clip range-handling differs 

480 # from torch.clamp (e.g. clip(int8, None, 128)). An in-range integer 

481 # bound (clip(int64 indices, 0, limit)) stays native and trace-safe. 

482 if ( 

483 self._needs_fallback(xt) 

484 or self._int_scalar_overflows(a_min, xt) 

485 or self._int_scalar_overflows(a_max, xt) 

486 ): 

487 return self._numpy_fallback(np.clip, x, a_min, a_max) 

488 x = xt 

489 a_min = self._clip_bound(a_min) 

490 a_max = self._clip_bound(a_max) 

491 # np.clip promotes by np.result_type(x, a_min, a_max): e.g. int input 

492 # with a float32 bound -> float32, int64 with a python float -> float64, 

493 # all-integer bounds keep the integer dtype. Cast x and any tensor bound 

494 # to that dtype; scalar bounds are coerced by torch.clamp. 

495 target = _np_result_dtype(x, a_min, a_max) 

496 if x.dtype != target: 

497 x = x.to(target) 

498 # torch.clamp rejects a Tensor bound paired with a Number bound; when 

499 # either bound is array-valued, tensorize both so np.clip's one-array- 

500 # one-scalar form (which torch.clamp otherwise rejects) works. 

501 tensorize = isinstance(a_min, torch.Tensor) or isinstance(a_max, torch.Tensor) 

502 a_min = self._cast_bound(a_min, target, tensorize) 

503 a_max = self._cast_bound(a_max, target, tensorize) 

504 return torch.clamp(x, min=a_min, max=a_max) 

505 

506 def _clip_bound(self, bound: Any) -> Any: 

507 """Pass scalar/``None`` bounds through; convert array-likes to tensors. 

508 

509 ``np.clip`` accepts any array-like bound, but ``torch.clamp`` needs a 

510 Number or Tensor, so list/tuple/ndarray bounds are converted. 

511 """ 

512 if bound is None or isinstance(bound, (int, float)): 

513 return bound 

514 return self.asarray(bound) 

515 

516 def _cast_bound(self, bound: Any, dtype: "torch.dtype", tensorize: bool) -> Any: 

517 """Cast a clip bound to ``dtype`` as a Tensor/scalar for ``torch.clamp``. 

518 

519 Tensor bounds are cast to ``dtype``; ``None`` passes through. Scalar 

520 bounds stay scalars for clamp's fast path unless ``tensorize`` (the other 

521 bound is a Tensor), in which case the scalar is broadcast to a Tensor so 

522 clamp does not reject a mixed Tensor/Number pair. 

523 """ 

524 if bound is None: 

525 return None 

526 if isinstance(bound, torch.Tensor): 

527 return bound if bound.dtype == dtype else bound.to(dtype) 

528 if tensorize: 

529 return torch.as_tensor(bound, dtype=dtype, device=self._device) 

530 return bound 

531 

532 def _promote_weak( 

533 self, x1: Any, x2: Any 

534 ) -> Tuple["torch.Tensor", "torch.Tensor"]: 

535 """Tensorize ``x1``/``x2`` at their joint ``np.result_type``, keeping 

536 Python scalars weak (NEP 50). 

537 

538 The dtype is computed from the *original* operands so a Python float/int 

539 contributes weak-scalar semantics: np.maximum(float32_array, 1.5) stays 

540 float32. Array operands promote strongly. Both are then materialized as 

541 tensors of that dtype for the binary op. 

542 """ 

543 target = _np_result_dtype(x1, x2) 

544 return self.asarray(x1).to(target), self.asarray(x2).to(target) 

545 

546 def _weak_int_overflow( 

547 self, 

548 x1: Any, 

549 x2: Any, 

550 t1: "torch.Tensor", 

551 t2: "torch.Tensor", 

552 ) -> bool: 

553 """True only when a weak Python int scalar is OUT OF RANGE for the 

554 integer array it is paired with. 

555 

556 NumPy range-checks the scalar against the array's dtype (raising on 

557 overflow), which Torch's silent wrap does not reproduce -- but only when 

558 the scalar actually overflows. An in-range scalar (e.g. ``maximum(int64 

559 indices, 0)`` or ``clip(indices, 0, limit)``) must stay on the native 

560 Torch path so it remains trace-safe; the float autograd path (a float 

561 array + scalar) is excluded entirely. 

562 """ 

563 return self._int_scalar_overflows(x1, t2) or self._int_scalar_overflows( 

564 x2, t1 

565 ) 

566 

567 @staticmethod 

568 def _int_scalar_overflows(scalar: Any, t: "torch.Tensor") -> bool: 

569 """True iff ``scalar`` is a Python int outside integer tensor ``t``'s 

570 dtype range (so NumPy would raise but Torch would silently wrap). A 

571 float/complex tensor or a non-int scalar never overflows here, keeping 

572 the native trace-safe path for in-range integer index operations.""" 

573 if not (isinstance(scalar, int) and not isinstance(scalar, bool)): 

574 return False 

575 if t.is_floating_point() or t.is_complex(): 

576 return False 

577 info = torch.iinfo(t.dtype) 

578 return not (info.min <= scalar <= info.max)

580 def _binary_minmax(self, np_func: Any, torch_func: Any, x1: Any, x2: Any) -> Any: 

581 t1, t2 = self.asarray(x1), self.asarray(x2) 

582 if self._needs_fallback(t1, t2) or self._weak_int_overflow(x1, x2, t1, t2): 

583 return self._numpy_fallback(np_func, x1, x2) 

584 return torch_func(*self._promote_weak(x1, x2)) 

585 

586 def maximum(self, x1: Any, x2: Any) -> Any: 

587 return self._binary_minmax(np.maximum, torch.maximum, x1, x2) 

588 

589 def minimum(self, x1: Any, x2: Any) -> Any: 

590 return self._binary_minmax(np.minimum, torch.minimum, x1, x2)

592 def where(self, condition: Any, x: Any, y: Any) -> Any: 

593 tx, ty = self.asarray(x), self.asarray(y) 

594 if self._needs_fallback(tx, ty) or self._weak_int_overflow(x, y, tx, ty): 

595 return self._numpy_fallback(np.where, condition, x, y) 

596 cond = self.asarray(condition) 

597 if cond.dtype is not torch.bool: 

598 cond = cond.to(torch.bool) 

599 # np.where promotes x/y (not the condition) by np.result_type with weak 

600 # Python scalars, so mixed int64/float32 values stay float64 like NumPy. 

601 return torch.where(cond, *self._promote_weak(x, y))

603 def cross(self, a: Any, b: Any) -> Any: 

604 ta, tb = self.asarray(a), self.asarray(b) 

605 # torch.linalg.cross rejects bool/unsigned/complex dtypes NumPy accepts. 

606 if self._needs_fallback(ta, tb): 

607 return self._numpy_fallback(np.cross, a, b) 

608 # np.cross promotes operands by np.result_type; cast both first so the 

609 # scalar-z and 3-vector results carry the NumPy dtype. 

610 a, b = self._promote_pair(ta, tb) 

611 la, lb = a.shape[-1], b.shape[-1] 

612 # np.cross of two 2-component vectors returns the scalar z-cross; 

613 # torch.linalg.cross requires length-3, so handle the 2D case directly. 

614 if la == 2 and lb == 2: 

615 return a[..., 0] * b[..., 1] - a[..., 1] * b[..., 0] 

616 # Mixed 2x3 / 3x2: np.cross treats the missing 3rd component as 0 and 

617 # returns a 3-vector; pad the 2-component operand with a zero z. 

618 if la == 2: 

619 a = self._pad_z(a) 

620 if lb == 2: 

621 b = self._pad_z(b) 

622 return torch.linalg.cross(a, b, dim=-1) 

623 

624 @staticmethod 

625 def _pad_z(v: "torch.Tensor") -> "torch.Tensor": 

626 """Append a zero third component along the last axis (2-vec -> 3-vec).""" 

627 zeros = torch.zeros( 

628 (*v.shape[:-1], 1), dtype=v.dtype, device=v.device 

629 ) 

630 return torch.cat((v, zeros), dim=-1)

632 def matmul(self, a: Any, b: Any) -> Any: 

633 ta, tb = self.asarray(a), self.asarray(b) 

634 # torch.matmul rejects bool/unsigned/complex-mixed dtypes NumPy accepts. 

635 if self._needs_fallback(ta, tb): 

636 return self._numpy_fallback(np.matmul, a, b) 

637 return torch.matmul(*self._promote_pair(ta, tb))

641 def _is_empty_axis(axis: Any) -> bool: 

642 """True for ``axis=()``/``[]``: NumPy reduces nothing, Torch reduces all.""" 

643 return isinstance(axis, (tuple, list)) and len(axis) == 0

645 def sum(self, x: Any, axis: Optional[Any] = None) -> Any: 

646 x = self.asarray(x) 

647 # bool/unsigned/complex accumulate differently in torch; delegate them. 

648 if self._needs_fallback(x): 

649 return self._numpy_fallback(np.sum, x, axis=axis) 

650 # np.sum applies an additive accumulator upcast on every path (e.g. 

651 # bool/int8 -> int64), unlike amax/amin. torch.sum upcasts integers to 

652 # int64, so cast the result to NumPy's accumulator dtype. 

653 target = _np_reduce_dtype(x) 

654 if self._is_empty_axis(axis): 

655 # axis=() does not reduce but still applies the accumulator upcast. 

656 return x if x.dtype == target else x.to(target) 

657 out = torch.sum(x) if axis is None else torch.sum(x, dim=axis) 

658 return out if out.dtype == target else out.to(target)

665 if self._is_empty_axis(axis): 

666 return x # np.amax(x, axis=()) returns x with its dtype preserved. 

667 return torch.amax(x) if axis is None else torch.amax(x, dim=axis)
